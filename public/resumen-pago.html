function configurarEventListeners() { const metodosDisponibles = document.querySelectorAll('.metodo-pago:not(.deshabilitado)'); metodosDisponibles.forEach(metodo => { metodo.addEventListener('click', function() { document.querySelectorAll('.metodo-pago').forEach(m => m.classList.remove('activo')); this.classList.add('activo'); metodoPagoSeleccionado = this.dataset.metodo; const botonesSeccion = document.getElementById('botonesSeccion'); const btnWebpay = document.getElementById('btnWebpay'); const btnOneclick = document.getElementById('btnOneclick'); if (botonesSeccion) botonesSeccion.style.display = 'block'; if (metodoPagoSeleccionado === 'webpay-plus') { if (btnWebpay) btnWebpay.style.display = 'block'; if (btnOneclick) btnOneclick.style.display = 'none'; } else if (metodoPagoSeleccionado === 'webpay-oneclick') { if (btnWebpay) btnWebpay.style.display = 'none'; if (btnOneclick) btnOneclick.style.display = 'block'; } }); }); // Botón Webpay Plus const btnWebpay = document.getElementById('btnWebpay'); if (btnWebpay) { btnWebpay.addEventListener('click', async function() { const aceptoTerminos = document.getElementById('aceptoTerminos'); if (!aceptoTerminos || !aceptoTerminos.checked) { alert('Debes marcar la casilla aceptando los términos y condiciones del contrato para continuar.'); aceptoTerminos.focus(); return; } try { const response = await fetch('https://creartransaccionwebpay-f4mre7bfoa-uc.a.run.app', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ordenId }) }); const data = await response.json(); if (data.redirect_url) { window.location.href = data.redirect_url; } else { alert('Error: ' + (data.error || 'No se pudo iniciar la transacción con Webpay')); } } catch (error) { console.error('Error:', error); alert('Error al conectar con la pasarela de pago. Por favor intenta de nuevo.'); } }); } // Botón Oneclick const btnOneclick = document.getElementById('btnOneclick'); if (btnOneclick) { btnOneclick.addEventListener('click', async function() { const aceptoTerminos = document.getElementById('aceptoTerminos'); if (!aceptoTerminos || !aceptoTerminos.checked) { alert('Debes marcar la casilla aceptando los términos y condiciones del contrato para continuar.'); aceptoTerminos.focus(); return; } try { const response = await fetch('https://iniciarinscripciononeclick-f4mre7bfoa-uc.a.run.app', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ordenId }) }); const data = await response.json(); if (data.redirect_url) { window.location.href = data.redirect_url; } else { alert('Error: ' + (data.error || 'No se pudo iniciar la inscripción con Oneclick')); } } catch (error) { console.error('Error:', error); alert('Error al procesar la solicitud con Oneclick. Por favor intenta de nuevo.'); } }); } }
